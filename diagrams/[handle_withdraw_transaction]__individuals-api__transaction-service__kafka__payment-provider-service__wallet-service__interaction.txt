@startuml

actor individuals_api_service
participant transaction_service
participant transaction_service_db
participant kafka

alt withdraw
	alt отправка транзакции withdraw в kafka
		individuals_api_service -> transaction_service : Отправить транзакцию [withdraw];
		transaction_service -> transaction_service_db : Сохранить транзакцию [withdraw] в состоянии [PENDING];
		transaction_service -> kafka : Отправить транзакцию [withdraw] в kafka, дождаться ответа результата отправки запроса;
		alt успешный ответ от kafka
			individuals_api_service <- transaction_service : Ответить клиенту сгенерированным UUID транзакции [withdraw];
		end
		alt ответ от kafka с ошибкой 
			transaction_service -> transaction_service_db : Отметить транзакцию [withdraw] в состояние [FAILED];
			individuals_api_service <- transaction_service : Ответить клиенту сообщением об ошибке в ходе обработки транзакции [withdraw];
		end
	end
	alt слушатель kafka сервисов-участников обработки транзакции withdraw
		transaction_service -> kafka : Получить сообщение об обработке транзакции [withdraw] от сервиса-участника;
		alt успешный ответ от kafka
			alt ответ от промежуточного участника оркестрации
				transaction_service -> kafka : Отправить транзакцию [withdraw] в kafka следующему участнику оркестрации, дождаться ответа результата отправки запроса;
				note right transaction_service
					В случае ошибки на любом из этапов обработки промежуточным сервисом,
					действовать в соответствии с инструкциями варианта [ответ с ошибкой от kafka].
				end note
				note right transaction_service
					Производить фиксацию успешно ответивших участников обработки транзакции.
					Если на каком-либо этапе обработки произойдет ошибка - данный список будет
					использоваться для произведения отмены транзакции.
				end note
			end
			alt ответ от конечного участника оркестрации
				transaction_service -> transaction_service_db : Отметить транзакцию [withdraw] в состояние [COMPLETED];
				transaction_service -> kafka : Отправить в kafka сообщение для webhook сервиса для последующего уведомления пользователя;
				note right transaction_service
					В случае ошибки от конечного участника оркестрации
					действовать в соответствии с инструкциями варианта [ответ с ошибкой от kafka].
				end note
			end
		end
		alt ответ с ошибкой от kafka
			transaction_service -> transaction_service_db : Отметить транзакцию [withdraw] в состояние [FAILED];
			transaction_service -> kafka : Отправить запрос на отмену транзакции [withdraw] в сервисы-участники оркестрации, которые уже приняли участие в обработке транзакции [withdraw];
			transaction_service -> kafka : Отправить в kafka сообщение для webhook сервиса для последующего уведомления пользователя;
		end
	end
end

@enduml
