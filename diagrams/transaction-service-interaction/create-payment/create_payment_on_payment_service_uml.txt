@startuml

actor individuals_api_service
participant transaction_service
participant transaction_service_db
participant payment_service
participant payment_provider_service
participant payment_provider_service_db

alt создание транзакции
	individuals_api_service -> transaction_service : Запрос [создание транзакции];
	transaction_service -> transaction_service : Преобразование запроса [создание транзакции] в запрос [создания платежа];
	transaction_service -> payment_service : Отправка rest-запроса [создание платежа];
	alt успешная обработка запроса [создание платежа] в сервисе [payment_service]
		payment_service -> payment_provider_service : Отправка rest-запроса [создание платежа];
		payment_provider_service -> payment_provider_service_db : Сохранение полученного запроса [создание платежа] в состоянии [PENDING], генерация ID платежа;
		alt успешная обработка запроса [создание платежа] в сервисе [payment_provider_service]
			payment_provider_service -> payment_service : ID созданного платежа;
			payment_service -> transaction_service : ID созданного платежа;
			transaction_service -> transaction_service : Сохранение ID платежа в сущности транзакции сервиса [transaction_service];
			transaction_service -> transaction_service_db : Сохранение сущности транзакции сервиса [transaction_service], генерация ID транзакции;
			transaction_service -> transaction_service : Оркестрация запроса [создание транзакции];
			transaction_service -> individuals_api_service : ID транзакции сервиса [transaction_service];
		else обработка запроса [создание платежа] в сервисе [payment_provider_service] с ошибкой
			payment_provider_service -> payment_service : Ошибка операции [создание платежа];
			payment_service -> transaction_service : Ошибка операции [создания платежа];
			transaction_service -> individuals_api_service : Уведомление пользователя об ошибке в процессе выполнения операции [создание транзакции];
		end
	else обработка запроса [создание платежа] в сервисе [payment_service] с ошибкой
		payment_service -> transaction_service : Ошибка операции [создания платежа];
		transaction_service -> individuals_api_service : Уведомление пользователя об ошибке в процессе выполнения операции [создание транзакции];
	end
end

@enduml
